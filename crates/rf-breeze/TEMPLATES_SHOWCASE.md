## Routes

### Auth Routes Preview
\`\`\`rust
pub fn auth_routes() -> Router {
    Router::new()
        // Guest routes (no auth required)
        .route("/login", get(show_login_form))
        .route("/login", post(login))
        .route("/register", get(show_register_form))
        .route("/register", post(register))
        
        // Protected routes (auth required)
        .route("/dashboard", get(show_dashboard))
        .route("/logout", post(logout))
        .layer(axum::middleware::from_fn(auth_layer))
}

## Middleware

### Auth Middleware Preview
\`\`\`rust
pub async fn auth_middleware(
    auth_user: Option<AuthUser<Claims>>,
    request: Request,
    next: Next,
) -> Result<Response, impl IntoResponse> {
    match auth_user {
        Some(_) => Ok(next.run(request).await),
        None => Err((StatusCode::UNAUTHORIZED, "Unauthorized")),
    }
}
\`\`\`

### Guest Middleware Preview
\`\`\`rust
pub async fn guest_middleware(
    auth_user: Option<AuthUser<Claims>>,
    request: Request,
    next: Next,
) -> Result<Response, impl IntoResponse> {
    match auth_user {
        Some(_) => Err(Redirect::to("/dashboard")),
        None => Ok(next.run(request).await),
    }
}

## Usage Examples

### Full Installation
\`\`\`rust
use rf_breeze::{BreezeScaffold, InstallOptions};

let breeze = BreezeScaffold::new(".")?;
breeze.install(&InstallOptions {
    with_api: true,
    with_email_verification: true,
    with_password_reset: true,
    output_dir: None,
}).await?;
\`\`\`

### Selective Installation
\`\`\`rust
// Install only what you need
breeze.install_views().await?;
breeze.install_controllers().await?;
breeze.install_routes().await?;
breeze.install_middleware().await?;
\`\`\`

## Integration Example

\`\`\`rust
use rf_breeze::BreezeScaffold;
use rf_auth::{JwtManager, PasswordHasher};
use rf_blade::BladeEngine;

// 1. Install scaffolding
let breeze = BreezeScaffold::new(".")?;
breeze.install(&InstallOptions::default()).await?;

// 2. Set up authentication
let jwt = Arc::new(JwtManager::new("secret")?);
let hasher = Arc::new(PasswordHasher::bcrypt(12)?);

// 3. Set up template engine
let blade = Arc::new(BladeEngine::new("resources/views")?);

// 4. Build application with generated routes
let app = Router::new()
    .merge(auth_routes())
    .layer(Extension(jwt))
    .layer(Extension(hasher))
    .layer(Extension(blade));
\`\`\`
