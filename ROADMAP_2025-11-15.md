```rust
// SOLL-ZUSTAND:
async fn load_has_many<R>(&self, db: &DatabaseConnection, foreign_key: &str)
    -> RelationshipResult<Vec<R>>
where
    R: EntityTrait + FromQueryResult,
{
    // 1. Get primary key value from parent model
    let parent_id = self.get_primary_key_value()?;

    // 2. Execute actual SeaORM query
    let results = R::Entity::find()
        .filter(Column::from(foreign_key).eq(parent_id))
        .all(db)
        .await?;

    Ok(results)
}
```

#### 1.2 BelongsTo Relationship
```rust
async fn load_belongs_to<R>(&self, db: &DatabaseConnection, foreign_key: &str)
    -> RelationshipResult<Option<R>>
where
    R: EntityTrait + FromQueryResult,
{
    // 1. Get foreign key value from this model
    let foreign_id = self.get_attribute(foreign_key)?;

    // 2. Find related model by ID
    let result = R::Entity::find_by_id(foreign_id)
        .one(db)
        .await?;

    Ok(result)
}
```

#### 1.3 BelongsToMany (Pivot Table)
```rust
async fn load_belongs_to_many<R>(&self,
    db: &DatabaseConnection,
    pivot_table: &str,
    foreign_pivot_key: &str,
    related_pivot_key: &str,
) -> RelationshipResult<Vec<R>>
{
    // 1. Get parent ID
    let parent_id = self.get_primary_key_value()?;

    // 2. Join through pivot table
    let results = R::Entity::find()
        .join(JoinType::InnerJoin, pivot_table)
        .filter(Column::from(foreign_pivot_key).eq(parent_id))
        .all(db)
        .await?;

    Ok(results)
}
```

#### 1.4 HasManyThrough
```rust
async fn load_has_many_through<R>(&self,
    db: &DatabaseConnection,
    through_model: &str,
    first_key: &str,
    second_key: &str,
```

#### 2.1 UniqueRule Implementation
```rust
pub struct UniqueRule {
    table: String,
    column: String,
    except_id: Option<i64>,  // Für Updates (exclude current record)
}

impl UniqueRule {
    pub fn new(table: impl Into<String>, column: impl Into<String>) -> Self {
        Self {
            table: table.into(),
            column: column.into(),
            except_id: None,
        }
    }

    pub fn except(mut self, id: i64) -> Self {
        self.except_id = Some(id);
        self
    }
}

#[async_trait]
impl ValidationRule for UniqueRule {
    async fn validate(&self, value: &Value, _data: &HashMap<String, Value>) -> RuleResult {
        let db = get_db_connection().await?;

        // Build query: SELECT COUNT(*) FROM {table} WHERE {column} = ?
        let mut query = Query::select()
            .column(Asterisk)
            .from(Iden::from(&self.table))
            .and_where(Expr::col(Iden::from(&self.column)).eq(value))
            .to_owned();

        // Exclude current record for updates
        if let Some(id) = self.except_id {
            query = query.and_where(Expr::col("id").ne(id)).to_owned();
        }

        let count: i64 = db.query_one(query).await?.try_get("", 0)?;

        if count > 0 {
            Err(format!("The {} has already been taken.", self.column))
        } else {
            Ok(())
        }
    }
}
```

#### 3.1 Single Relation Eager Loading
```rust
async fn load_relation<M>(&self, models: &mut Vec<M>, relation: &EagerLoadRelation)
    -> EagerLoadResult<()>
where
    M: ModelTrait + HasRelationship,
{
    if models.is_empty() {
        return Ok(());
    }

    // 1. Extract foreign key values from parent models
    let parent_ids: Vec<i64> = models.iter()
        .map(|m| m.get_primary_key_value())
        .collect();

    // 2. Load ALL related models in ONE query using IN clause
    // Instead of: SELECT * FROM posts WHERE user_id = 1
    //             SELECT * FROM posts WHERE user_id = 2
    //             ... (N queries)
    // We do:      SELECT * FROM posts WHERE user_id IN (1, 2, 3, ...) (1 query!)
    let related_models = relation.related_entity
        .find()
        .filter(Expr::col(&relation.foreign_key).is_in(parent_ids))
        .all(self.db)
        .await?;

    // 3. Group related models by foreign key
    let mut grouped: HashMap<i64, Vec<RelatedModel>> = HashMap::new();
    for model in related_models {
        let foreign_key_value = model.get_foreign_key_value(&relation.foreign_key);
        grouped.entry(foreign_key_value)
            .or_insert_with(Vec::new)
            .push(model);
    }

    // 4. Attach related models to parent models
    for parent in models {
        let parent_id = parent.get_primary_key_value();
        let related = grouped.remove(&parent_id).unwrap_or_default();
        parent.set_relation(&relation.name, related);
    }

    Ok(())
}
```

#### 3.2 Nested Eager Loading
```rust
// User::with("posts.comments").get()
// Should load: users, posts for those users, comments for those posts

async fn load_nested_relations<M>(&self, models: &mut Vec<M>, path: &str)
    -> EagerLoadResult<()>
{
    let parts: Vec<&str> = path.split('.').collect();

    // Load first level (posts)
    self.load_relation(models, &EagerLoadRelation::new(parts[0])).await?;

    // Load second level (comments) if exists
    if parts.len() > 1 {
        // Get all posts
        let all_posts: Vec<Post> = models.iter()
            .flat_map(|m| m.get_relation::<Post>(parts[0]))
            .collect();

        // Load comments for all posts
        self.load_relation(&mut all_posts, &EagerLoadRelation::new(parts[1])).await?;
    }

    Ok(())
}
```

#### 3.3 Query Optimization
```rust
impl<M: ModelTrait> EagerLoader<M> {
    pub fn with(&mut self, relation: &str) -> &mut Self {
        self.relations.push(relation.to_string());
        self
    }

    pub fn with_count(&mut self, relation: &str) -> &mut Self {
        // Add COUNT subquery instead of loading full models
        self.counts.push(relation.to_string());
        self
    }

    pub async fn get(&self) -> EagerLoadResult<Vec<M>> {
        // 1. Load base models
        let mut models = M::Entity::find().all(self.db).await?;

        // 2. Load each relation
        for relation_name in &self.relations {
            self.load_relation(&mut models, relation_name).await?;
        }

        // 3. Add counts
        for count_name in &self.counts {
            self.add_count(&mut models, count_name).await?;
        }

        Ok(models)
    }
}
```
